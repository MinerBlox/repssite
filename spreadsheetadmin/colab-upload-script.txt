def slugify(value):
    value = value.strip().lower()
    value = re.sub(r"[^a-z0-9\s-]", "", value)
    value = re.sub(r"\s+", "-", value)
    value = re.sub(r"-+", "-", value)
    return value.strip("-")


def extension_for(filename):
    mime_type, _ = mimetypes.guess_type(filename)
    extension = mimetypes.guess_extension(mime_type or "") or ""
    if not extension and "." in filename:
        extension = "." + filename.rsplit(".", 1)[-1].lower()
    if extension == ".jpe": extension = ".jpg"
    return extension or ".png"


def upload_image_to_r2(local_filename, object_key):
    mime_type, _ = mimetypes.guess_type(local_filename)
    with open(local_filename, "rb") as image_file:
        s3.put_object(
            Bucket=R2_BUCKET,
            Key=object_key,
            Body=image_file,
            ContentType=mime_type or "application/octet-stream",
            CacheControl="public, max-age=31536000, immutable"
        )
    public_key = "/".join(quote(part, safe="") for part in object_key.split("/"))
    return f"{R2_PUBLIC_BASE}/{public_key}"


def compact_index_item(product_id, data):
    return {
        "id": str(product_id),
        "name": str(data.get("name") or ""),
        "category": str(data.get("category") or ""),
        "brand": str(data.get("brand") or ""),
        "sortOrder": int(data.get("sortOrder") or 0),
        "isActive": data.get("isActive") is not False,
    }


def save_catalog_index(items):
    items = sorted(items, key=lambda item: int(item.get("sortOrder") or 0))
    payload = {
        "version": int(time.time() * 1000),
        "updatedAt": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "items": items,
    }
    raw = json.dumps(payload, ensure_ascii=False, separators=(",", ":")).encode("utf-8")
    s3.put_object(
        Bucket=R2_BUCKET,
        Key=CATALOG_INDEX_KEY,
        Body=raw,
        ContentType="application/json; charset=utf-8",
        CacheControl="no-cache"
    )
    return payload


def build_catalog_index_from_firestore():
    print("\nBuilding persistent search/category index...")
    items = []
    for product_doc in db.collection("liveproducts").stream():
        data = product_doc.to_dict() or {}
        items.append(compact_index_item(product_doc.id, data))
    payload = save_catalog_index(items)
    print(f"✅ Catalog index ready: {len(payload['items']):,} items")
    print(f"R2 object: {CATALOG_INDEX_KEY}")
    return payload


def load_or_build_catalog_index():
    try:
        obj = s3.get_object(Bucket=R2_BUCKET, Key=CATALOG_INDEX_KEY)
        payload = json.loads(obj["Body"].read().decode("utf-8"))
        if isinstance(payload.get("items"), list):
            print(f"✅ Existing catalog index loaded: {len(payload['items']):,} items")
            return payload
    except Exception:
        pass
    return build_catalog_index_from_firestore()


def upsert_catalog_index_item(payload, product_id, data):
    by_id = {str(item.get("id")): item for item in payload.get("items", []) if item.get("id")}
    by_id[str(product_id)] = compact_index_item(product_id, data)
    new_payload = save_catalog_index(list(by_id.values()))
    payload.clear()
    payload.update(new_payload)


def existing_categories():
    try:
        names = []
        for category_doc in db.collection("categories").stream():
            data = category_doc.to_dict() or {}
            name = str(data.get("name") or category_doc.id).strip()
            if name and name.lower() != "uncategorised": names.append(name)
        return sorted(set(names), key=str.lower)
    except Exception:
        return []

    image_filename = next(iter(uploaded_image.keys()))

    try:
        image_url = upload_image_to_r2(image_filename, object_key)

            "name": item_name,
            "price": yuan_price,
            "currency": "CNY",
            "category": category,
            "brand": "",
            "badge": "",
            "imageUrl": image_url,
            "productUrl": "",
            "agentUrl": agent_link,
            "description": "",

            "isActive": True,
            "isOurPick": False,
            "homeSummer": False,
            "homeAutumn": False,

            category_slug = slugify(category)

                    categories.append(category)
                    categories.sort(key=str.lower)

        print("\n✅ Done.")
        print("Firestore collection: liveproducts")
        print(f"Product document ID: {product_id}")

        print(f"Image URL: {image_url}")
        print("✅ Search/category index updated too.")

    except Exception as error:
        print("\n❌ Something went wrong with this item:")
        print(error)
